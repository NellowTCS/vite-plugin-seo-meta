Config Reference

SeoPluginOptions

{
  site?: SiteConfig;
  pages?: Record<string, PageMeta>;
  robots?: "auto" | boolean | RobotsFileOptions;
  sitemap?: "auto" | boolean | SitemapOptions;
  validate?: "warn" | "error" | false;
  configFile?: string | false;
}

SiteConfig

{
  url: string;
  name?: string;
  title?: string;
  titleTemplate?: string;
  description?: string;
  image?: string;
  imageAlt?: string;
  twitter?: string | TwitterInfo;
  locale?: string;
  lang?: string;
  author?: string;
  keywords?: string | string[];
  robotsByUserAgent?: Record<string, string | RobotsDirectives>;
  favicon?: FaviconConfig;
  openGraph?: OpenGraphInfo;
  jsonLd?: JsonLdConfig;
}

twitter accepts a bare @handle as shorthand for twitter:site.

PageMeta

{
  title?: string;
  description?: string;
  canonical?: string;
  lang?: string;
  image?: string;
  imageAlt?: string;
  keywords?: string | string[];
  author?: string;
  noindex?: boolean;
  robots?: string | RobotsDirectives;
  robotsByUserAgent?: Record<string, string | RobotsDirectives>;
  openGraph?: OpenGraphInfo;
  twitter?: TwitterInfo;
  jsonLd?: JsonLdConfig;
  alternates?: HreflangAlternate[];
  sitemap?: boolean;
  priority?: number;
  changefreq?: SitemapChangefreq;
  lastmod?: string | Date;
  meta?: Record<string, string>;
  links?: Array<{ rel: string; href: string } & Record<string, string>>;
}

FaviconConfig

{
  icon?: string;
  appleTouchIcon?: string;
  appleTouchIconPrecomposed?: string;
  manifest?: string;
  maskIcon?: string;
  maskIconColor?: string;
  themeColor?: string | ThemeColorEntry[];
  mobileWebApp?: MobileWebAppConfig;
  msapplication?: MsTileConfig;
}

MobileWebAppConfig

{
  title?: string;
  capable?: boolean;
  statusBarStyle?: "default" | "black" | "black-translucent" | "transparent";
}

MsTileConfig

{
  tileColor?: string;
  tileImage?: string;
  config?: string;
}

RobotsFileOptions

{
  userAgent?: string | string[];
  disallow?: string[];
  allow?: string[];
  groups?: RobotsGroup[];
  rules?: string[];
  sitemap?: string | false;
  merge?: boolean;
  disallowNoindex?: boolean;
  host?: string;
  crawlDelay?: number;
  cleanParam?: string | string[];
}

RobotsGroup

{
  userAgent: string | string[];
  disallow?: string[];
  allow?: string[];
  crawlDelay?: number;
  cleanParam?: string | string[];
}

SitemapOptions

{
  filename?: string;
  exclude?: string[];
  defaultChangefreq?: SitemapChangefreq;
  defaultPriority?: number;
  siteUrl?: string;
  maxUrls?: number;
}

SitemapChangefreq

"always" | "hourly" | "daily" | "weekly" | "monthly" | "yearly" | "never"

ImageRef / ImageInput

// ImageRef
{
  src: string;
  alt?: string;
  width?: number;
  height?: number;
  type?: string;
  secureUrl?: string;
  thumbnail?: string;
}

// ImageInput: string | ImageRef | { src?, url?, href?, alt?, width?, height?, type?, secureUrl?, secure_url?, thumbnail? }

thumbnail is the video sitemap thumbnail URL. Required for video entries.

TwitterInfo

{
  card?: TwitterCardType;
  title?: string;
  description?: string;
  site?: TwitterHandle;
  siteId?: string;
  creator?: TwitterHandle;
  creatorId?: string;
  image?: string;
  imageAlt?: string;
  imageWidth?: number;
  imageHeight?: number;
  label1?: string;
  data1?: string;
  label2?: string;
  data2?: string;
  player?: TwitterPlayer;
  apps?: TwitterApps;
}

TwitterPlayer

{
  url: string;
  width: number;
  height: number;
  stream?: string;
}

TwitterApps

{
  iphone?: TwitterAppEntry;
  ipad?: TwitterAppEntry;
  googleplay?: TwitterAppEntry;
  country?: string;
}

TwitterAppEntry

{
  name?: string;
  id?: string;
  url?: string;
}

TwitterCardType

"summary" | "summary_large_image" | "player" | "app"

JsonLdConfig

{
  organization?: boolean | OrganizationJsonLd;
  website?: boolean | WebSiteJsonLd;
  article?: boolean | ArticleJsonLd;
  product?: boolean | ProductJsonLd;
  extra?: Record<string, unknown>[];
}

HreflangAlternate

{
  hreflang: string;
  href: string;
}

RobotsDirectives

{
  noindex?: boolean;
  nofollow?: boolean;
  none?: boolean;
  all?: boolean;
  nosnippet?: boolean;
  noarchive?: boolean;
  noimageindex?: boolean;
  notranslate?: boolean;
  indexifembedded?: boolean;
  nositelinkssearchbox?: boolean;
  maxSnippet?: number;
  maxImagePreview?: "none" | "standard" | "large";
  maxVideoPreview?: number;
  unavailableAfter?: string | Date;
}
